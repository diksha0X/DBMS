CREATE TABLE Customer (
    Cust_id VARCHAR2(5) PRIMARY KEY
        CHECK (Cust_id LIKE 'A%'),
    Fname VARCHAR2(30) NOT NULL,
    Lname VARCHAR2(30) NOT NULL,
    Area VARCHAR2(5),
    Phone NUMBER(10)
);

CREATE TABLE Movie (
    Mv_no NUMBER(5) PRIMARY KEY,
    Title VARCHAR2(50) NOT NULL,
    Type VARCHAR2(20) NOT NULL,
    Star VARCHAR2(10) NOT NULL,
    Price NUMBER(8,2) NOT NULL
);

CREATE TABLE Invoice (
    Inv_no VARCHAR2(5) PRIMARY KEY
        CHECK (Inv_no LIKE 'I%'),
    Mv_no NUMBER(5),
    Cust_id VARCHAR2(5),
    Issue_date DATE NOT NULL,
    Return_date DATE NOT NULL,
    FOREIGN KEY (Mv_no) REFERENCES Movie(Mv_no),
    FOREIGN KEY (Cust_id) REFERENCES Customer(Cust_id)
);

SELECT E.EMPLOYEE_ID,
       E.FIRST_NAME || ' ' || E.LAST_NAME AS EMPLOYEE_NAME,
       E.MANAGER_ID,
       M.FIRST_NAME || ' ' || M.LAST_NAME AS MANAGER_NAME

LEFT JOIN EMP M
ON E.MANAGER_ID = M.EMPLOYEE_ID;

       M2.FIRST_NAME || ' ' || M2.LAST_NAME AS MANAGERS_MANAGER
