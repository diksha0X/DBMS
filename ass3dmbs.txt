1. Create EMPLOYEE table
CREATE TABLE EMPLOYEE (
    EMPLOYEE_ID NUMBER(6),
    LAST_NAME VARCHAR2(25),
    JOB_ID VARCHAR2(10),
    SALARY NUMBER(8,2),
    COMM_PCT NUMBER(4,2),
    MGR_ID NUMBER(6),
    DEPARTMENT_ID NUMBER(4)
);

3. Display last_name, job_id, employee_id, with employee_id first
SELECT employee_id, last_name, job_id
FROM EMPLOYEE;


4. Employees of department 60
SELECT *
FROM EMPLOYEE
WHERE department_id = 60;

5. Employee whose last name is King
SELECT *
FROM EMPLOYEE
WHERE last_name = 'King';

6. Display unique job_id with alias JOB_TITLE
SELECT DISTINCT job_id AS JOB_TITLE
FROM EMPLOYEE;

7. Display last name, salary and salary increased by Rs.300
SELECT last_name, salary, salary + 300 AS "Increased Salary"
FROM EMPLOYEE;

8. Annual compensation + one-time bonus of Rs.100
SELECT last_name,
       salary,
       (salary * 12) + 100 AS "Annual Compensation"
FROM EMPLOYEE;

9. Employees who get commission
SELECT *
FROM EMPLOYEE
WHERE comm_pct IS NOT NULL;

10. Employees who do not get commission
SELECT *
FROM EMPLOYEE
WHERE comm_pct IS NULL;

11. Employee ID, Department ID and Salary where salary > 5000
SELECT employee_id, department_id, salary
FROM EMPLOYEE
WHERE salary > 5000;

12. Last name and salary between 4000 and 7000
SELECT last_name, salary
FROM EMPLOYEE
WHERE salary BETWEEN 4000 AND 7000;

13. Employees whose salary is 6000, 6500 or 7000
SELECT *
FROM EMPLOYEE
WHERE salary IN (6000, 6500, 7000);

14. Employees in departments 10, 20, 30 or 50
SELECT *
FROM EMPLOYEE
WHERE department_id IN (10, 20, 30, 50);

15. Employees whose salary is not 5000
SELECT *
FROM EMPLOYEE
WHERE salary <> 5000;

16. All CLERKS
SELECT *
FROM EMPLOYEE
WHERE job_id LIKE '%CLERK%';

17. Change job_id to Grade_A where salary > 5000
UPDATE EMPLOYEE
SET job_id = 'Grade_A'
WHERE salary > 5000;

FROM EMPLOYEE
