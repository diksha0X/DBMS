1.
SELECT LOWER(FIRST_NAME) AS FIRST_NAME
FROM EMPLOYEE_DETAILS
WHERE DEPARTMENT_ID IN (20, 50, 80);


2.
SELECT UPPER(LAST_NAME) AS LAST_NAME
FROM EMPLOYEE_DETAILS
WHERE DEPARTMENT_ID IN (10, 110, 60);


3.
SELECT INITCAP(JOB_ID) AS JOB_ID
FROM EMPLOYEE_DETAILS;


4.
SELECT CONCAT(FIRST_NAME, CONCAT(' ', LAST_NAME)) AS NAME
FROM EMPLOYEE_DETAILS;


5.
SELECT LAST_NAME, LENGTH(LAST_NAME) AS NAME_LENGTH
FROM EMPLOYEE_DETAILS
WHERE COMMISSION_PCT IS NOT NULL;


6.
SELECT FIRST_NAME, INSTR(FIRST_NAME, 'a') AS POSITION
FROM EMPLOYEE_DETAILS;


7.
SELECT CONCAT(FIRST_NAME, CONCAT(' ', LAST_NAME)) AS NAME,
LENGTH(LAST_NAME) AS NAME_LENGTH,
INSTR(LAST_NAME, 'a') AS POSITION
FROM EMPLOYEE_DETAILS
WHERE JOB_ID LIKE '%CLERK%';


8.
SELECT LPAD(SALARY, 10, '$') AS SALARY
FROM EMPLOYEE_DETAILS;


9.
SELECT TRIM('e' FROM FIRST_NAME) AS TRIMMED
FROM EMPLOYEE_DETAILS
WHERE FIRST_NAME LIKE '%e%';


10.
SELECT DEPARTMENT_ID, SALARY, HIRE_DATE
FROM EMPLOYEE_DETAILS
WHERE LENGTH(FIRST_NAME) = 6;


11.
SELECT JOB_ID, INSTR(FIRST_NAME, 'e') AS POSITION
FROM EMPLOYEE_DETAILS
WHERE FIRST_NAME LIKE '%e%';


12.
SELECT LAST_NAME
FROM EMPLOYEE_DETAILS
WHERE INSTR(JOB_ID, 'REP') = 4;


13.
SELECT LAST_NAME, MOD(SALARY, 3000) AS REMAINDER
FROM EMPLOYEE_DETAILS
WHERE JOB_ID = 'SA_REP';


14.
SELECT TO_CHAR(SALARY, '$9,999.00') AS SALARY
FROM EMPLOYEE_DETAILS;


15.
SELECT TO_CHAR(SALARY, '$9999') AS SALARY
FROM EMPLOYEE_DETAILS
WHERE DEPARTMENT_ID IN (60, 90, 110);


16.
SELECT LAST_NAME,
NVL(TO_CHAR(COMMISSION_PCT), 'No Commission') AS COMM
FROM EMPLOYEE_DETAILS;


17.
SELECT LAST_NAME,
ROUND(SYSDATE - HIRE_DATE) AS DAYS_ELAPSED
FROM EMPLOYEE_DETAILS;


18.
SELECT LAST_NAME,
ROUND((SYSDATE - HIRE_DATE) / 7) AS WEEKS_ELAPSED
FROM EMPLOYEE_DETAILS;


19.
SELECT LAST_NAME,
ROUND(MONTHS_BETWEEN(SYSDATE, HIRE_DATE) / 12) AS YEARS_ELAPSED
FROM EMPLOYEE_DETAILS;


20.
SELECT FIRST_NAME,
ROUND(MONTHS_BETWEEN(SYSDATE, HIRE_DATE)) AS MONTHS_WORKED
FROM EMPLOYEE_DETAILS;


21.
SELECT FIRST_NAME,
LAST_DAY(HIRE_DATE) AS LAST_DAY
FROM EMPLOYEE_DETAILS;


22.
SELECT FIRST_NAME,
TO_CHAR(HIRE_DATE, 'YEAR') AS HIRE_YEAR
FROM EMPLOYEE_DETAILS;


23.
SELECT ADD_MONTHS(SYSDATE, 5) AS "FUTURE DATE"
FROM DUAL;


24.
SELECT FIRST_NAME || ' ' || LAST_NAME AS NAME,
TO_CHAR(HIRE_DATE, 'DAY') AS DAY,
TO_CHAR(HIRE_DATE, 'DD-MON-YYYY') AS DATE_OF_JOINING
FROM EMPLOYEE_DETAILS;


25.
SELECT FIRST_NAME,
(SALARY * 12) + NVL(COMMISSION_PCT, 0) AS ANNUAL_SALARY,
ROUND((SALARY * 12) + NVL(COMMISSION_PCT, 0), 2) AS ROUNDED_SALARY
FROM EMPLOYEE_DETAILS;
