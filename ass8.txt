CREATE TABLE Sports_Stud (
ROLL NUMBER(5) PRIMARY KEY,
NAME VARCHAR2(30) UNIQUE,
DEPT VARCHAR2(5) CHECK (DEPT IN ('IT','CS')),
SPORTS_FEE NUMBER(8) CHECK (SPORTS_FEE >= 1000)
);

CREATE TABLE Cultural_Stud (
ROLL NUMBER(5) PRIMARY KEY,
NAME VARCHAR2(30) UNIQUE,
DEPT VARCHAR2(5) CHECK (DEPT IN ('IT','CS')),
CUL_FEES NUMBER(8) CHECK (CUL_FEES >= 600)
);

SELECT NAME,
       TRUNC(MONTHS_BETWEEN(SYSDATE,BIRTHDAY)/12) AS AGE,
       TO_CHAR(BIRTHDAY,'MONTH') AS MONTH_OF_BIRTH
FROM Sports_Stud
UNION
SELECT NAME,
       TRUNC(MONTHS_BETWEEN(SYSDATE,BIRTHDAY)/12),
       TO_CHAR(BIRTHDAY,'MONTH')
FROM Cultural_Stud;


10.

FROM Sports_Stud
UNION
SELECT LAST_DAY(BIRTHDAY)
FROM Cultural_Stud;
