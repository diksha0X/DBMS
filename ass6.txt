2.
SELECT FNAME AS "Employee Name"
FROM EMP
ORDER BY FNAME ASC;


3.
SELECT FNAME AS "Employee Name"
FROM EMP
ORDER BY FNAME DESC;


4.
SELECT HIRE_DATE
FROM EMP
ORDER BY HIRE_DATE ASC;


5.
SELECT *
FROM EMP
WHERE FNAME LIKE 'J%'
OR FNAME LIKE 'M%'
ORDER BY LNAME;


6.
SELECT MAX(SAL) AS Max,
MIN(SAL) AS Min,
AVG(SAL) AS Avg,
SUM(SAL) AS Sum
FROM EMP;


7.
SELECT JOB_ID,
MAX(SAL) AS Max,
MIN(SAL) AS Min,
AVG(SAL) AS Avg,
SUM(SAL) AS Sum
FROM EMP
GROUP BY JOB_ID;


8.
SELECT JOB_ID, COUNT(*) AS Total_Employees
FROM EMP
GROUP BY JOB_ID;


9.
SELECT COUNT(*) AS Total_Managers
FROM EMP
WHERE JOB_ID LIKE '%MGR';


10.
SELECT MAX(SAL) - MIN(SAL) AS Salary_Difference
FROM EMP;


11.
SELECT MAX(SAL) AS Max_Salary,
AVG(SAL) AS Avg_Salary
FROM EMP
WHERE JOB_ID = 'IT_PROG';


12.
SELECT MIN(FNAME) AS First_Name,
MAX(FNAME) AS Last_Name
FROM EMP;


13.
SELECT MIN(HIRE_DATE) AS "First Hire_Date",
MAX(HIRE_DATE) AS "Last Hire_Date"
FROM EMP;


14.
SELECT MAX(SAL) AS Max_Salary,
AVG(SAL) AS Avg_Salary
FROM EMP
WHERE JOB_ID LIKE '%CLERK%';


15.
SELECT DEPT_ID,
MIN(SAL) AS Minimum_Salary
FROM EMP
GROUP BY DEPT_ID;


16.
SELECT DEPT_ID,
MIN(SAL) AS Minimum_Salary
FROM EMP
GROUP BY DEPT_ID
HAVING MIN(SAL) > 3000
ORDER BY Minimum_Salary DESC;


17.
SELECT FNAME || ' whose designation is ' ||
JOB_ID || ' gets ' || SAL ||
' but wants to earn ' || (3*SAL) AS DETAILS
FROM EMP;


18.
SELECT E_ID || ', ' || FNAME || ', ' ||
LNAME || ', ' ||
TO_CHAR(HIRE_DATE,'DD-MON-RR') || ', ' ||
JOB_ID || ', ' || SAL || ', ' ||
DEPT_ID AS EMPLOYEE_DETAILS
FROM EMP;


19.
SELECT SYSDATE AS TODAY
FROM DUAL;


20.

TO_CHAR(HIRE_DATE,'DD') AS DAY,
TO_CHAR(HIRE_DATE,'YYYY') AS YEAR

21.

TO_CHAR(HIRE_DATE,'DD-MONTH-YY') AS HIRE_DATE
FROM EMP;


22.

TO_CHAR(HIRE_DATE,'MONTH') AS HIRE_MONTH
FROM EMP;


23.
SELECT FNAME, E_ID,
TO_CHAR(HIRE_DATE,'DD-MON-YYYY') AS HIRE_DATE
FROM EMP;


24.
SELECT FNAME, E_ID,
TO_CHAR(HIRE_DATE,'MONTH-DD-YYYY') AS HIRE_DATE
FROM EMP;


25.
SELECT TO_CHAR(SYSDATE,'YEAR') AS YEAR_IN_WORDS
FROM DUAL;


26.
SELECT SYSDATE AS TODAY,
SYSDATE + 15 AS AFTER_15_DAYS,
SYSDATE - 15 AS BEFORE_15_DAYS
FROM DUAL;
